# ============================================
# CONFIGURACIÓN COMPLETA PARA SW1
# ============================================

configure terminal
ip ssh version 2
ip domain name local.com
hostname SW1
enable secret cisco

# Deshabilitar logging a consola
no logging console



# Habilitar CEF
ip cef


vlan 10
vlan 20
vlan 90
vlan 120
 name blackhole
exit
username cisco privilege 15 secret cisco
crypto key generate rsa modulus 2048

# ===== INTERFACES =====

interface Ethernet0/0
 switchport access vlan 10
 switchport mode access
 switchport port-security
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 15
exit

# Ethernet0/1 - VLAN 20 (Acceso)
interface Ethernet0/1
 switchport access vlan 20
 switchport mode access
 switchport port-security
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 15
exit

# Ethernet0/2 - Trunk (VLAN 10,20,90)
interface Ethernet0/2
 switchport trunk encapsulation dot1q
 switchport mode trunk 
 switchport trunk native vlan 90
exit

# Ethernet0/3 - Management
interface Ethernet0/3
 no switchport 
 description to port3.sandbox-backend
 ip address 10.10.20.173 255.255.255.0
exit

# Ethernet1/0 - VLAN 120 (Blackhole)
interface range Ethernet1/0-3
switchport mode access
 switchport access vlan 120
 shutdown
exit



# ===== LÍNEAS =====
line con 0
 exec-timeout 0 0
 logging synchronous
  password cisco

exit

line vty 0 4
 login local
 transport input ssh

exit

# ===== SPANNING TREE =====
spanning-tree mode rapid-pvst
spanning-tree extend system-id
service password-encryption
copy running-config startup-config
