!================PC-1===============

su
cat << 'EOF' > /usr/share/udhcpc/default6.script
#!/bin/sh
case "$1" in
  deconfig)
    ip -6 addr flush dev "$interface" scope global
    ;;
  bound|renew)
    [ -n "$ipv6" ] && ip -6 addr replace "$ipv6/64" dev "$interface"
    ip -6 route replace default via fe80::10 dev "$interface" 2>/dev/null || true
    ;;
esac
EOF
chmod +x /usr/share/udhcpc/default6.script
killall udhcpc udhcpc6 2>/dev/null || true

udhcpc -i eth0 -b
udhcpc6 -i eth0 -s /usr/share/udhcpc/default6.script -b
ifconfig
! =====================================================================
