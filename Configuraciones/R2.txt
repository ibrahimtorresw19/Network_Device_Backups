# ============================================
# CONFIGURACIÓN COMPLETA PARA R2
# ============================================



configure terminal
hostname R2
enable secret cisco 
ipv6 unicast-routing
no logging console
ip cef

ip dhcp excluded-address 192.168.30.1 192.168.30.10
ip dhcp excluded-address 192.168.40.1 192.168.40.10

ip dhcp pool VLAN30_A
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 8.8.8.8
 lease 7
exit

ip dhcp pool VLAN40_A
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 8.8.8.8
 lease 7
exit

# ===== DHCP IPv6 =====
ipv6 dhcp pool VLAN30_EMPRESA
 address prefix 2001:db8:acad:30::/64 lifetime infinite infinite
 dns-server 2001:db8:acad:30::A
exit

ipv6 dhcp pool VLAN40_EMPRESA
 address prefix 2001:db8:acad:40::/64 lifetime infinite infinite
 dns-server 2001:db8:acad:40::A
exit

# ===== USUARIOS =====
enable secret cisco
username cisco privilege 15 secret cisco
crypto key generate rsa modulus 2048

# ===== INTERFACES =====
interface Ethernet0/0
 no ip address
exit

interface Ethernet0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip ospf 10 area 0
 ipv6 address 2001:db8:acad:30::1/64
 ipv6 address fe80::30 link-local
 ipv6 nd prefix 2001:db8:acad:30::/64
 ipv6 nd managed-config-flag
 ipv6 nd other-config-flag
 ipv6 dhcp server VLAN30_EMPRESA
 ipv6 ospf 10 area 0
exit

interface Ethernet0/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip ospf 10 area 0
 ipv6 address 2001:db8:acad:40::1/64
 ipv6 address fe80::40 link-local
 ipv6 nd prefix 2001:db8:acad:40::/64
 ipv6 nd managed-config-flag
 ipv6 nd other-config-flag
 ipv6 dhcp server VLAN40_EMPRESA
 ipv6 ospf 10 area 0
exit

interface Ethernet0/1
 ip address 1.1.1.2 255.255.255.0
 ip ospf 10 area 0
 ipv6 address 2001:db8:acad:1::2/64
 ipv6 address fe80::2 link-local
 ipv6 ospf 10 area 0
exit

interface Ethernet0/2
 description to port1.sandbox-backend
 vrf forwarding Mgmt-intf
 ip address 10.10.20.172 255.255.255.0
exit

interface Ethernet0/3
 no ip address
 shutdown
exit


# ===== OSPF =====
router ospf 10
 router-id 2.2.2.2
 network 1.1.1.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.40.0 0.0.0.255 area 0
exit

ipv6 router ospf 10
 router-id 2.2.2.2
exit

# ===== LÍNEAS =====
spanning-tree mode rapid-pvst

line con 0
 exec-timeout 0 0
 password cisco
 logging synchronous
exit

line vty 0 4
 exec-timeout 0 0
 password cisco
 login local
 transport input ssh
exit

service password-encryption
end
copy running-config startup-config
