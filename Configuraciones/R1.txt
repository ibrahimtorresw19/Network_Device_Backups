configure terminal
hostname R1
enable secret cisco
ip domain name local.com
no logging console
ip cef
ipv6 unicast-routing

! ===== USUARIOS Y SSH =====
username cisco privilege 15 secret cisco
crypto key generate rsa modulus 2048
ip ssh version 2

! ===== INTERFAZ SIMULADA DE INTERNET =====
interface Loopback100
 description SIMULACION_ISP_INTERNET
 ip address 200.1.1.1 255.255.255.252
 ip nat outside
 ipv6 address 2001:db8:ffff::1/64
exit

! ===== DHCP IPv4 =====
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10

ip dhcp pool VLAN10_A
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
 lease 7
exit

ip dhcp pool VLAN20_A
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8
 lease 7
exit

! ===== DHCP IPv6 =====
ipv6 dhcp pool VLAN10_EMPRESA
 address prefix 2001:db8:acad:10::/64 lifetime infinite infinite
 dns-server 2606:4700:4700::1111
exit

ipv6 dhcp pool VLAN20_EMPRESA
 address prefix 2001:db8:acad:20::/64 lifetime infinite infinite
 dns-server 2606:4700:4700::1111
exit

! ===== ACLs IPv4 =====

ip access-list extended ACL_VLAN10_IN
 permit udp any eq bootpc any eq bootps
 permit ip 192.168.10.0 0.0.0.255 host 192.168.10.1
 permit ip 192.168.10.0 0.0.0.255 192.168.40.0 0.0.0.255
 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.10.0 0.0.0.255 any
exit


ip access-list extended ACL_VLAN20_IN
 permit udp any eq bootpc any eq bootps
 permit ip 192.168.20.0 0.0.0.255 host 192.168.20.1
 deny ip 192.168.20.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.20.0 0.0.0.255 any
exit

! ===== ACLs IPv6 =====

ipv6 access-list ACL6_VLAN10_IN
 permit icmp any any
 permit udp fe80::/10 any eq 547
 permit ipv6 2001:db8:acad:10::/64 host 2001:db8:acad:10::1
 permit ipv6 2001:db8:acad:10::/64 2001:db8:acad:40::/64
 deny ipv6 2001:db8:acad:10::/64 2001:db8:acad::/48
 permit ipv6 2001:db8:acad:10::/64 any
exit

ipv6 access-list ACL6_VLAN20_IN
 permit icmp any any
 permit udp fe80::/10 any eq 547
 permit ipv6 2001:db8:acad:20::/64 host 2001:db8:acad:20::1
 deny ipv6 2001:db8:acad:20::/64 2001:db8:acad::/48
 permit ipv6 2001:db8:acad:20::/64 any
exit

! ===== INTERFACES Y SUBINTERFACES =====
interface Ethernet0/0
 description TRUNK_sW1
 no ip address
 no shutdown
exit

interface Ethernet0/0.10
 description red_vlan10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
 ip access-group ACL_VLAN10_IN in
 ip ospf 10 area 0
 ipv6 address 2001:db8:acad:10::1/64
 ipv6 address fe80::10 link-local
 ipv6 nd prefix 2001:db8:acad:10::/64 2592000 604800 no-autoconfig
 ipv6 nd managed-config-flag
 ipv6 nd other-config-flag
 ipv6 dhcp server VLAN10_EMPRESA
 ipv6 traffic-filter ACL6_VLAN10_IN in
 ipv6 ospf 10 area 0
exit

interface Ethernet0/0.20
 description red_vlan20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
 ip access-group ACL_VLAN20_IN in
 ip ospf 10 area 0
 ipv6 address 2001:db8:acad:20::1/64
 ipv6 address fe80::20 link-local
 ipv6 nd prefix 2001:db8:acad:20::/64 2592000 604800 no-autoconfig
 ipv6 nd managed-config-flag
 ipv6 nd other-config-flag
 ipv6 dhcp server VLAN20_EMPRESA
 ipv6 traffic-filter ACL6_VLAN20_IN in
 ipv6 ospf 10 area 0
exit

interface Ethernet0/0.90
 description vlan_nativa
 encapsulation dot1Q 90 native
exit

interface Ethernet0/1
 description enlace_R1_R2
 ip address 10.0.12.1 255.255.255.252
 ip ospf network point-to-point
 ip ospf 10 area 0
 ipv6 address 2001:db8:acad:1::1/64
 ipv6 address fe80::1 link-local
 ipv6 ospf network point-to-point
 ipv6 ospf 10 area 0
 ip nat inside
 no shutdown
exit

interface Ethernet0/3
 no ip address
 shutdown
exit

! ===== PROCESOS OSPF =====
router ospf 10
 router-id 1.1.1.1
 passive-interface Ethernet0/0.10
 passive-interface Ethernet0/0.20
 default-information originate always
exit

ipv6 router ospf 10
 router-id 1.1.1.1
 passive-interface Ethernet0/0.10
 passive-interface Ethernet0/0.20
 default-information originate always
exit

! ===== RUTAS =====
ip route 0.0.0.0 0.0.0.0 Loopback100
ipv6 route ::/0 Loopback100

! ===== NAT / PAT =====
ip nat pool PUBLICOS 200.1.1.5 200.1.1.10 netmask 255.255.255.252

ip access-list extended NAT_LAN
 permit ip 192.168.10.0 0.0.0.255 any
 permit ip 192.168.20.0 0.0.0.255 any
 permit ip 192.168.30.0 0.0.0.255 any
 permit ip 192.168.40.0 0.0.0.255 any
exit

ip nat inside source list NAT_LAN pool PUBLICOS overload
ip nat inside source static tcp 192.168.30.5 80 200.1.1.11 80
ip nat inside source static tcp 192.168.40.5 80 200.1.1.11 8080

 password cisco
 logging synchronous
exit

 exec-timeout 0 0

exit

end
